## Componente con hooks (useState)

import { useState } from "react"

export function Counter() {
  const [n, setN] = useState(0);
  return (
    <div style={{ display: "flex", alignItems: "center", gap: 8, margin: "12px 0" }}>
      <button onClick={() => setN(n - 1)} style={{ padding: "4px 10px" }}>−</button>
      <span style={{ fontVariantNumeric: "tabular-nums", minWidth: 24, textAlign: "center" }}>{n}</span>
      <button onClick={() => setN(n + 1)} style={{ padding: "4px 10px" }}>+</button>
    </div>
  );
}

<Counter />
